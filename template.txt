function translateText(s){
  if (Object.prototype.hasOwnProperty.call(EN,s)) return EN[s];
  let t=s;
  t=t.replace(/^Bonjour, Dr /,'Hello, Dr ').replace(/^Bonjour, /,'Hello, ');
  t=t.replace(/^Seuil : (\d+) mmHg$/,'Threshold: $1 mmHg')
    .replace(/^(\d+) entrées$/,'$1 entries').replace(/^(\d+) signalé\(s\)$/,'$1 reported')
    .replace(/Intensité (\d+)\/(\d+)/g,'Intensity $1/$2');
  for (const [fr,en] of Object.entries(EN)) {
    if (fr==='Fiche :' || fr==='Dernière mesure :' || fr==='• sélectionné') t=t.replaceAll(fr,en);
  }
  return t;
}
